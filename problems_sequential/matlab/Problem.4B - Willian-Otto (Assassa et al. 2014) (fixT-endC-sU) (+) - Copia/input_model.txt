%--------------------------------------------------------------------------

par(3)={'Vj = 2000'}; %

% algebraic equations
 y(1) = 9.11e13*exp(-29508/(Rg*x(8)));

dx(6) = {'2*y(2)*x(2)^3 + y(3)*x(4)*x(2) + x(2)*y(7)*(x(6)+x(7)) + y(6)*x(2)*x(6) - (y(7)*x(2) + y(9)*(x(5) + x(6) + x(7) ) + y(7)*x(2) + y(8)*x(3) )*x(6)'};

dx(7) = {'y(6)*x(5)*x(2) +  x(6)*x(2)*x(7) - (y(7)*x(2) + y(9)*(x(5) + x(6) + x(7) ) + y(7)*x(2) + y(8)*x(3) )*x(7)'};
